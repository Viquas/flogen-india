3. Create `webgen/lib/claim-tracking.ts` — the fire-and-forget tracking utility. Follow the exact pattern from `lib/ai/cost-tracker.ts` (try/catch wrapper, error logged not thrown):
   ```typescript
   import { createAdminClient } from '@/lib/supabase/admin'

   export type ClaimEventType =
     | 'preview_view'
     | 'claim_page_view'
     | 'cta_click'
     | 'plan_selected'
     | 'payment_initiated'
     | 'payment_completed'
     | 'customization_submitted'

   interface TrackEventParams {
     siteSlug: string
     eventType: ClaimEventType
     ip?: string | null
     userAgent?: string | null
     metadata?: Record<string, unknown>
   }

   /**
    * Track a claim funnel event. Fire-and-forget safe — errors are logged, not thrown.
    * Use for server-side tracking (SSR pages, webhooks, server actions).
    */
   export async function trackClaimEvent(params: TrackEventParams): Promise<void> {
     try {
       const supabase = createAdminClient()
       const { error } = await supabase.from('claim_events').insert({
         site_slug: params.siteSlug,
         event_type: params.eventType,
         ip: params.ip ?? null,
         user_agent: params.userAgent ?? null,
         metadata: params.metadata ?? {},
       })
       if (error) {
         console.error('[ClaimTracking] Failed to track event:', error.message)
       }
     } catch (err) {
       console.error('[ClaimTracking] Unexpected error tracking event:', err)
     }
   }
   ```

4. Create `webgen/app/api/analytics/claim-event/route.ts` — POST endpoint for client-side event logging. This is needed because client components (claim-page-client.tsx) cannot call server-only tracking directly:
   ```typescript
   import { trackClaimEvent, type ClaimEventType } from '@/lib/claim-tracking'
   import { headers } from 'next/headers'

   const VALID_EVENTS: ClaimEventType[] = [
     'preview_view', 'claim_page_view', 'cta_click',
     'plan_selected', 'payment_initiated', 'payment_completed',
     'customization_submitted',
   ]

   export async function POST(request: Request) {
     try {
       const body = await request.json()
       const { siteSlug, eventType, metadata } = body

       if (!siteSlug || !eventType || !VALID_EVENTS.includes(eventType)) {
         return Response.json({ error: 'Invalid event' }, { status: 400 })
       }

       const headersList = await headers()
       const ip = headersList.get('x-forwarded-for')?.split(',')[0]?.trim() || null
       const userAgent = headersList.get('user-agent') || null

       // Fire-and-forget — do not await in response path (but we do await since it's fast)
       await trackClaimEvent({ siteSlug, eventType, ip, userAgent, metadata })

       return Response.json({ ok: true })
     } catch {
       return Response.json({ error: 'Failed' }, { status: 500 })
     }
   }
   ```

IMPORTANT: Do NOT use `export const dynamic` or any route segment config on this route — it is a POST handler and dynamic by default.
