Data fetching (use createAdminClient for service-role queries):
```typescript
const admin = createAdminClient()
// Fetch most recent paid claim for this user
const { data: claim } = await admin
  .from('claims')
  .select('id, project_id, plan, status, client_name, client_email')
  .eq('auth_user_id', user.id)
  .in('status', ['paid', 'customizing', 'completed'])
  .order('created_at', { ascending: false })
  .limit(1)
  .single()

if (!claim) redirect('/portal/login')  // No valid claim -- shouldn't happen, but handle gracefully

// Fetch project data
const { data: project } = await admin
  .from('projects')
  .select('id, business_data, generated_code, slug, version')
  .eq('id', claim.project_id)
  .single()

// Count pending and in-progress requests for status
const { count: pendingRequests } = await admin
  .from('client_requests')
  .select('id', { count: 'exact', head: true })
  .eq('auth_user_id', user.id)
  .eq('status', 'pending')

const { count: inProgressRequests } = await admin
  .from('client_requests')
  .select('id', { count: 'exact', head: true })
  .eq('auth_user_id', user.id)
  .eq('status', 'in_progress')
```
